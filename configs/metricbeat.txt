настройка мониторинга и сборка метрик с cерверов кластера ES через metricbeat
echo "deb [trusted=yes] https://artifacts.elastic.co/packages/8.x/apt stable main" | sudo tee /etc/apt/sources.list.d/elastic-8.x.list
apt update
apt-get install metricbeat

посмотрим какие метрики он умеет собирать
metricbeat modules list

Включаем модуль для эластика
metricbeat modules enable elasticsearch-xpack

nano /etc/metricbeat/modules.d/elasticsearch-xpack.yml 

- module: elasticsearch
  xpack.enabled: true
  period: 10s
  hosts: ["https://localhost:9200"]
  ssl.certificate_authorities: ["/etc/elasticsearch/certs/http_ca.crt"]
  username: "elastic"
  password: "123456"


nano /etc/metricbeat/metricbeat.yml

output.elasticsearch:
  hosts: ["localhost:9200"]
  ssl.certificate_authorities: ["/etc/elasticsearch/certs/http_ca.crt"]
  preset: balanced
  protocol: "https"
  username: "elastic"
  password: "123456"


systemctl restart metricbeat
systemctl enable metricbeat
systemctl status metricbeat


Проверим подключение к Elasticsearch
metricbeat test output

Посмотреть есть ли индекс
curl -k -u elastic:123456 https://10.20.30.41:9200/_cat/indices/metricbeat-*?v

journalctl -u metricbeat -f

Проверяем в кибане 
http://192.168.56.16:5601/app/monitoring#/overview
