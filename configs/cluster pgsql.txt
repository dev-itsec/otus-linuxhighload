### установка 
Binary download
https://developer.hashicorp.com/consul/install
wget https://releases.hashicorp.com/consul/1.22.6/consul_1.22.6_linux_amd64.zip
unzip consul_1.22.6_linux_amd64.zip -d /usr/bin/
consul -v

### настройка
useradd -r -c 'Consul' consul
mkdir -p /var/lib/consul /etc/consul.d
chown consul:consul /var/lib/consul /etc/consul.d
chmod 775 /var/lib/consul /etc/consul.d

nano /etc/systemd/system/consul.service

[Unit]
Description=Consul Service Discovery Agent
Documentation=https://www.consul.io/
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=consul
Group=consul
ExecStart=/usr/bin/consul agent -config-dir=/etc/consul.d
ExecReload=/bin/kill -HUP $MAINPID
KillSignal=SIGINT
TimeoutStopSec=5
Restart=on-failure
SyslogIdentifier=consul

[Install]
WantedBy=multi-user.target

### Установка

wget --quiet -O - https://www.postgresql.org/media/keys/ACCC4CF8.asc | sudo gpg --dearmor -o /usr/share/keyrings/postgresql-archive-keyring.gpg
echo "deb [signed-by=/usr/share/keyrings/postgresql-archive-keyring.gpg] http://apt.postgresql.org/pub/repos/apt $(lsb_release -cs)-pgdg main" | sudo tee /etc/apt/sources.list.d/pgdg.list
apt update

apt install -y postgresql-16 postgresql-contrib-16

Подготавливаем PostgreSQL для работы с сервисом patroni.
Узнаем путь к базам PostgreSQL.
sudo -u postgres psql -c 'SHOW data_directory;'

data_directory = /var/lib/postgresql/16/main

Останавливаем и убираем с автозапуска PostgreSQL и удаляем базы на всех серверах.
Patroni будет управлять PostgreSQL, поэтому мы отключаем дефолтную службу PostgreSQL
sudo systemctl stop postgresql
sudo systemctl disable postgresql
rm -rf /var/lib/postgresql/16/main/*

### Созадем базы данных и проверяем их репликацию
Создаем базу на лидер ноде

sudo -u postgres psql
\l
CREATE DATABASE cms;

# Проверить, что db-node-2 действительно в режиме реплики
sudo -u postgres psql -c "SELECT pg_is_in_recovery();"
# Должно вернуть 't' (true) - значит это реплика

Пробуем создать базу на реплике db-node-2 и не получается
postgres=# CREATE DATABASE cms;
ERROR:  cannot execute CREATE DATABASE in a read-only transaction

Бывает, что один из участников кластера подвисает и репликация может работать неправильно.
В этом случае, можно выполнить повторную инициализацию узла кластера
patronictl -c /etc/patroni/patroni.yml reinit pg-sql

Смена лидера (мастера ноды)
patronictl -c /etc/patroni/patroni.yml failover pg-sql

Увидеть подробный лог работы patroni 
journalctl -u patroni -f

### Создаем базу данных и пользователя для Grafana

sudo -u postgres psql

-- Создаем базу данных для Grafana
CREATE DATABASE grafana;

-- Создаем пользователя 
CREATE USER grafana_user WITH PASSWORD 'grafana123';

-- Даем права на подключение к базе
GRANT CONNECT ON DATABASE grafana TO grafana_user;

-- Даем права на создание таблиц (Grafana сама создаст схему)
GRANT CREATE, TEMP ON DATABASE grafana TO grafana_user;

-- Переключаемся на базу и даем права на схему public
\c grafana

GRANT ALL PRIVILEGES ON SCHEMA public TO grafana_user;
GRANT ALL PRIVILEGES ON ALL TABLES IN SCHEMA public TO grafana_user;
ALTER DEFAULT PRIVILEGES IN SCHEMA public GRANT ALL ON TABLES TO grafana_user;
