##############################################

nano /etc/elasticsearch/elasticsearch.yml

везде нужно прописать 
discovery.seed_hosts: ["10.20.30.51:9300","10.20.30.52:9300","10.20.30.53:9300"]
http.host: 0.0.0.0   # чтобы работали запросы на https://localhost:9200

и на мастере закомментировать
cluster.initial_master_nodes: ["10.20.30.51"]
