###################### Filebeat Configuration Example #########################

# ============================== Filebeat inputs ===============================

filebeat.inputs:

# Input для системных логов
- type: filestream
  id: system-logs
  enabled: true
  paths:
    - /var/log/*.log
    - /var/log/secure
    - /var/log/messages
    - /var/log/syslog

# Input для логов Docker контейнеров
- type: filestream
  id: docker-logs
  enabled: true
  paths:
    - /var/lib/docker/containers/*/*.log
  parsers:
    - ndjson:
        target: docker
        add_error_key: true
  processors:
    # Добавляем метаданные Docker (контейнер, образ, etc.)
    - add_docker_metadata:
        host: "unix:///var/run/docker.sock"
        match_fields:
          - "container.id"
    # Переименовываем поле log в message
    - rename:
        fields:
          - from: "docker.log"
            to: "message"
        ignore_missing: true
    # Добавляем поле для фильтрации
    - add_fields:
        target: ""
        fields:
          service.type: "docker"

# ============================== Filebeat modules ==============================

filebeat.config.modules:
  path: ${path.config}/modules.d/*.yml
  reload.enabled: false

# ======================= Elasticsearch template setting =======================

setup.template.settings:
  index.number_of_shards: 1

# ================================== Outputs ===================================

# ------------------------------ Logstash Output -------------------------------
output.logstash:
  hosts: ["10.20.30.54:5044", "10.20.30.55:5044"]
  loadbalance: true
  timeout: 30
  ssl.certificate_verification: false
  ssl.enabled: false

  # Если нужен SSL

# ================================= Processors =================================

processors:
  - add_host_metadata:
      when.not.contains.tags: forwarded
  - add_cloud_metadata: ~
  - add_docker_metadata: ~
  - add_kubernetes_metadata: ~

# ================================== Logging ===================================

logging.level: info

Запуск и проверка Filebeat

# Проверить конфигурацию

# Проверить статус
