Итак, у нас есть кластер Elastic из трех нод

es-node1 https://10.20.30.51:9200
es-node2 https://10.20.30.52:9200
es-node3 https://10.20.30.53:9200


Нужно настроить кибану на них на серверах web-node1\web-node2

если нет докера там, то поставить докер с компос

curl -fsSL https://get.docker.com -o get-docker.sh
sh get-docker.sh
sudo usermod -aG docker $USER

Создать роль и пользователя для Kibana
curl -k -u elastic:123456 -X POST "https://10.20.30.51:9200/_security/role/full_access_role" -H 'Content-Type: application/json' -d'
{
  "cluster": ["all"],
  "indices": [
    {
      "names": ["*"],
      "privileges": ["all"],
      "allow_restricted_indices": true
    }
  ],
  "run_as": ["*"]
}'

curl -k -u elastic:123456 -X POST "https://10.20.30.51:9200/_security/user/kibana_user" -H 'Content-Type: application/json' -d'
{
  "password" : "kibana123",
  "roles" : ["kibana_admin", "monitoring_user", "full_access_role"],
  "full_name" : "Kibana User"
}'

# корнервой из любой ноды эластика cat /etc/elasticsearch/certs/http_ca.crt

# запускаем контейнер кибаны
cd /opt/kibana
nano docker-compose.yml

services:
  kibana:
    image: docker.elastic.co/kibana/kibana:8.19.13
    container_name: kibana
    volumes:
      - /opt/kibana/certs:/usr/share/kibana/config/certs:ro
    environment:
      - SERVER_NAME=kibana
      - SERVER_HOST=0.0.0.0
      - SERVER_PUBLICBASEURL=http://192.168.56.100:5601 # IP HAProxy
      - ELASTICSEARCH_HOSTS=["https://10.20.30.51:9200","https://10.20.30.52:9200","https://10.20.30.53:9200"]
      - ELASTICSEARCH_USERNAME=kibana_user
      - ELASTICSEARCH_PASSWORD=kibana123
      - ELASTICSEARCH_SSL_CERTIFICATEAUTHORITIES=/usr/share/kibana/config/certs/http_ca.crt
    ports:
      - "5601:5601"
    restart: unless-stopped

docker compose up -d
